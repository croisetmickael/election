function createAllSheets() {
  const SHEET_ID = "17tnj8SS68OLEO_2JqXfvjO9nSzXYLuvPgVX8AgxyT3Q";
  const ss = SpreadsheetApp.openById(SHEET_ID);
  
  // Supprimer la feuille par défaut
  const defaultSheet = ss.getSheetByName("Feuille1");
  if (defaultSheet) ss.deleteSheet(defaultSheet);
  
  // Créer les 5 feuilles
  createSheet(ss, "Résultats", "#003D82");
  createSheet(ss, "Officiers", "#003D82");
  createSheet(ss, "Non-Officiers", "#E31C23");
  createSheet(ss, "PATS", "#009B9B");
  
  // Résumé
  let resume = ss.insertSheet("Résumé");
  resume.appendRow(["Collège", "Nombre de votes"]);
  resume.appendRow(["Officiers", 0]);
  resume.appendRow(["Non-Officiers", 0]);
  resume.appendRow(["PATS", 0]);
  resume.getRange("A1:B1").setBackground("#666666").setFontColor("white").setFontWeight("bold");
  
  SpreadsheetApp.flush();
}

function createSheet(ss, name, color) {
  let sheet = ss.insertSheet(name);
  sheet.appendRow([
    "Timestamp", "Prénom", "Nom", "IP",
    "Priorité 1", "Priorité 2", "Priorité 3", "Priorité 4", "Priorité 5", "Priorité 6"
  ]);
  sheet.getRange("A1:J1").setBackground(color).setFontColor("white").setFontWeight("bold");
}
